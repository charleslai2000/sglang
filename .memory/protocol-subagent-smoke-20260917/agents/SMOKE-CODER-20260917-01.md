### Validator source (complete, no elisions)

```python
#!/usr/bin/env python3
"""Deterministic stdlib-only validator for protocol result artifacts.

Checks, per artifact:
  (a) all 7 required section headings are present as Markdown ATX headings
      with the exact required text (case-sensitive);
  (b) the "Terminal state" section contains exactly one standalone terminal
      token among DONE / BLOCKED / FAILED.

Operational definition used for (b): the terminal section is the text from the
"Terminal state" heading to the next ATX heading (or end of file); standalone
token means a regex word-boundary match of exactly one of the three tokens.
Whole-document token occurrences are printed for information only and do not
affect the verdict.

Exit code: 0 if every artifact PASSes, 1 if any artifact FAILs, 2 on usage error.
"""

import re
import sys

REQUIRED_HEADINGS = [
    "Task",
    "Scope",
    "Result",
    "Evidence",
    "Defects-blockers",
    "Remaining claims-exclusions",
    "Terminal state",
]

TERMINAL_TOKENS = ("DONE", "BLOCKED", "FAILED")
TOKEN_RE = re.compile(r"\b(DONE|BLOCKED|FAILED)\b")
TERMINAL_HEADING_RE = re.compile(r"^#{1,6}[ \t]+Terminal state[ \t]*$", re.MULTILINE)
ANY_HEADING_RE = re.compile(r"^#{1,6}[ \t]+.+?[ \t]*$", re.MULTILINE)


def heading_re(name):
    return re.compile(r"^#{1,6}[ \t]+" + re.escape(name) + r"[ \t]*$", re.MULTILINE)


def terminal_section(text):
    match = TERMINAL_HEADING_RE.search(text)
    if match is None:
        return None
    rest = text[match.end():]
    nxt = ANY_HEADING_RE.search(rest)
    if nxt is not None:
        rest = rest[:nxt.start()]
    return rest


def validate(path):
    with open(path, "r", encoding="utf-8") as handle:
        text = handle.read()

    missing = [h for h in REQUIRED_HEADINGS if not heading_re(h).search(text)]
    headings_pass = not missing

    section = terminal_section(text)
    tokens = [] if section is None else TOKEN_RE.findall(section)
    terminal_pass = len(tokens) == 1

    return {
        "path": path,
        "headings_pass": headings_pass,
        "missing": missing,
        "tokens": tokens,
        "terminal_pass": terminal_pass,
        "doc_tokens": TOKEN_RE.findall(text),
    }


def main(argv):
    if len(argv) < 2:
        print("usage: python3 - <artifact> [<artifact> ...]", file=sys.stderr)
        return 2

    results = [validate(p) for p in argv[1:]]
    all_pass = True
    for r in results:
        headings_n = len(REQUIRED_HEADINGS) - len(r["missing"])
        token_text = ", ".join(r["tokens"]) if r["tokens"] else "none"
        verdict = "PASS" if (r["headings_pass"] and r["terminal_pass"]) else "FAIL"
        all_pass = all_pass and verdict == "PASS"
        print("artifact: " + r["path"])
        print("  (a) required headings: %s (%d/7 present)" % (
            "PASS" if r["headings_pass"] else "FAIL", headings_n))
        if r["missing"]:
            print("      missing: " + ", ".join(r["missing"]))
        print("  (b) exactly one terminal token: %s (terminal-section tokens: %s)" % (
            "PASS" if r["terminal_pass"] else "FAIL", token_text))
        print("      whole-document token occurrences (informational): " +
              (", ".join(r["doc_tokens"]) if r["doc_tokens"] else "none"))
        print("  verdict: " + verdict)
        print()

    print("overall: " + ("PASS" if all_pass else "FAIL"))
    return 0 if all_pass else 1


if __name__ == "__main__":
    sys.exit(main(sys.argv))
```
